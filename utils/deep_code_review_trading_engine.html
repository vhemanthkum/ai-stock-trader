Stage 1A — hard elimination gates (add to _score_batch)

def _passes_hard_gates(self, info: dict, hist: pd.DataFrame) -> tuple[bool, str]:
    """
    Hard boolean gates. Any fail → stock eliminated immediately.
    Run cheapest checks first to avoid unnecessary computation.
    
    UNCERTAINTY NOTE: yfinance field names like 'marketCap',
    'averageVolume', 'debtToEquity' — verify these exist in
    yf.Ticker(symbol).fast_info or .info for your version of yfinance.
    Field names have changed across yfinance versions.
    """
    # Gate 1: Price above ₹5 (removes extreme penny stocks)
    close_prices = hist["Close"].dropna()
    if close_prices.empty or float(close_prices.iloc[-1]) < 5.0:
        return False, "price_below_5"

    # Gate 2: Minimum liquidity — 20-day avg volume > 50,000 shares
    # (₹50 stock × 50K shares = ₹25L minimum daily liquidity)
    volume = hist["Volume"].dropna()
    avg_vol = float(volume.tail(20).mean()) if len(volume) >= 5 else 0
    if avg_vol < 50_000:
        return False, "illiquid"

    # Gate 3: Sufficient data (newly listed stocks are unreliable)
    if len(close_prices) < 20:
        return False, "insufficient_history"

    # Gate 4: Not in catastrophic downtrend
    # If price is below BOTH SMA-20 AND down >30% in 21 days: skip
    sma20 = float(close_prices.tail(20).mean())
    price_21d_ago = float(close_prices.iloc[-21]) if len(close_prices) >= 21 else float(close_prices.iloc[0])
    drawdown = (float(close_prices.iloc[-1]) - price_21d_ago) / price_21d_ago * 100
    if float(close_prices.iloc[-1]) < sma20 and drawdown < -25:
        return False, "catastrophic_downtrend"

    # NOTE: Ideally add promoter pledge and debt gates here.
    # However, yfinance does not reliably provide these for Indian stocks.
    # These should be fetched from screener_parser.py fundamentals
    # and applied as a post-scoring filter, not here in the batch loop.
    # Worth verifying what info['debtToEquity'] returns for NSE tickers.

    return True, "ok"

Upgraded _score_ticker — institutional weights

def _score_ticker_v2(self, data, yf_ticker: str) -> tuple[float, str]:
    """
    Institutional scoring matrix.
    Returns (score, reason_string).
    Score range: roughly -60 to +120.
    Stocks scoring below 15 after gates are weak setups.
    """
    score = 0.0
    reasons = []

    # --- Extract data safely ---
    try:
        if isinstance(data.columns, pd.MultiIndex):
            close  = data["Close"][yf_ticker].dropna()
            volume = data["Volume"][yf_ticker].dropna()
            high   = data["High"][yf_ticker].dropna()
            low    = data["Low"][yf_ticker].dropna()
        else:
            close  = data["Close"].dropna()
            volume = data["Volume"].dropna()
            high   = data["High"].dropna()
            low    = data["Low"].dropna()
        if len(close) < 5:
            return 0.0, "insufficient_data"
    except (KeyError, AttributeError):
        return 0.0, "data_error"

    latest = float(close.iloc[-1])
    prev   = float(close.iloc[-2]) if len(close) >= 2 else latest
    pct_chg = ((latest - prev) / prev * 100) if prev > 0 else 0

    # Moving averages (use available data, don't fake 200-DMA)
    sma5  = float(close.tail(5).mean())
    sma20 = float(close.tail(20).mean()) if len(close) >= 20 else sma5

    # 52-week high proxy (use all available data in the download window)
    # NOTE: With 30d download, this is a 30-day high, not 52-week.
    # The full 52-week high comes from yfinance_collector.py (1y period).
    # Label it accurately.
    recent_high = float(close.max())
    pct_from_high = ((latest - recent_high) / recent_high * 100) if recent_high > 0 else 0

    # Volume
    today_vol  = float(volume.iloc[-1]) if len(volume) > 0 else 0
    avg_vol_20 = float(volume.tail(20).mean()) if len(volume) >= 5 else today_vol
    vol_ratio  = (today_vol / avg_vol_20) if avg_vol_20 > 0 else 1.0

    # === MOMENTUM BLOCK (40% of score weight conceptually) ===

    # 1. Proximity to period high (breakout setup)
    if pct_from_high >= -2.0:
        score += 35
        reasons.append("Near period high")
    elif pct_from_high >= -5.0:
        score += 20
        reasons.append("Close to period high")

    # 2. Trend structure (200-DMA not available in 30d; use SMA-20 as proxy)
    if latest > sma20:
        score += 20
        reasons.append("Above SMA-20")
    else:
        score -= 10

    if latest > sma5 and latest > sma20:
        score += 10
        reasons.append("Short-term uptrend")
    elif latest < sma5 and latest < sma20:
        score -= 20
        reasons.append("Short-term downtrend")

    # 3. Day's price action
    if 1.5 <= pct_chg <= 5.0:  # Healthy move, not parabolic
        score += 15
        reasons.append(f"Healthy move +{pct_chg:.1f}%")
    elif pct_chg > 5.0:
        score += 5   # Parabolic — late entry risk, smaller bonus
        reasons.append(f"Strong move +{pct_chg:.1f}% (late entry risk)")
    elif pct_chg < -4.0:
        score += 10  # Potential bounce, but needs fundamental check
        reasons.append(f"Oversold drop {pct_chg:.1f}% (bounce candidate)")

    # === VOLUME BLOCK ===
    # Accumulation zone: 1.2x-2.5x is healthy; >3x often panic/dump
    if 1.2 <= vol_ratio <= 2.5:
        score += 25
        reasons.append(f"Accumulation vol {vol_ratio:.1f}x")
    elif vol_ratio > 2.5:
        score += 10  # Spike — could be distribution, less reliable
        reasons.append(f"Vol spike {vol_ratio:.1f}x (verify direction)")
    elif vol_ratio < 0.5:
        score -= 15  # Drying up, no interest
        reasons.append("Vol drying up")

    return score, " | ".join(reasons) if reasons else "no_signal"

Root cause of your 429 failures — flat sleep is not rate limiting
# YOUR CODE (main.py line 350):

# THIS DOES NOT PREVENT 429s. Here's why:
# 1. You run 3 agents IN PARALLEL (asyncio.gather) per stock.
#    Those 3 parallel calls go to Groq simultaneously.
#    Groq's free limit is 30 RPM. 3 calls × 15 stocks = 45 calls
#    to Groq minimum. If gather fires them at t=0, t=0, t=0
#    and the next stock fires at t=4.0, t=4.0, t=4.0:
#    In minute 1: 45 calls. Groq limit 30/min → 429 on call 31.
#
# 2. You have TWO separate places making Groq calls:
#    - fundamental_agent → Groq (llama-3.3-70b-versatile)
#    - technical_agent   → Groq (llama-3.1-8b-instant)
#    - news_agent        → Groq (llama-3.1-8b-instant)
#    All three in the same asyncio.gather() call.
#    The 4-second sleep between stocks doesn't help —
#    the 3 calls per stock already burst before the sleep.
# FIX: Per-provider semaphore with token bucket
import asyncio
from collections import deque
import time

class ProviderRateLimiter:
    """Token bucket rate limiter per provider."""
    def __init__(self, rpm: int, rpd: int):
        self.rpm = rpm
        self.rpd = rpd
        self.minute_window: deque = deque()
        self.day_total = 0
        self._lock = asyncio.Lock()

    async def acquire(self, provider_name: str):
        async with self._lock:
            now = time.monotonic()
            # Evict entries older than 60 seconds
            while self.minute_window and now - self.minute_window[0] > 60:
                self.minute_window.popleft()

            if len(self.minute_window) >= self.rpm:
                # Must wait until oldest entry expires
                wait_sec = 60 - (now - self.minute_window[0]) + 0.1
                logger.info(f"{provider_name}: RPM limit, waiting {wait_sec:.1f}s")
                await asyncio.sleep(wait_sec)
                # Re-evict after sleep
                now = time.monotonic()
                while self.minute_window and now - self.minute_window[0] > 60:
                    self.minute_window.popleft()

            if self.day_total >= self.rpd:
                raise RuntimeError(f"{provider_name}: Daily limit {self.rpd} reached. Use fallback.")

            self.minute_window.append(now)
            self.day_total += 1

# One instance per provider, at module level:
GROQ_LIMITER   = ProviderRateLimiter(rpm=30, rpd=1000)
GOOGLE_LIMITER = ProviderRateLimiter(rpm=15, rpd=1500)
